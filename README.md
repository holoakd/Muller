function mostrar(nombre) {

    var secciones =
    document.getElementsByClassName("seccion");

    for (var i = 0; i < secciones.length; i++) {

        secciones[i].style.display = "none";

    }

    document.getElementById(nombre).style.display = "block";

    window.scrollTo(0, 0);

}
